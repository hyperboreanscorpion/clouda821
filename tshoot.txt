def get_log_excerpt(
    log_file: Path,
    max_lines: int = 15,
) -> str:

    try:
        with log_file.open(
            "r",
            encoding="utf-8",
            errors="replace",
        ) as fh:

            lines = deque(
                fh,
                maxlen=max_lines,
            )

        return "".join(lines)[-2500:]

    except Exception:
        return "Unable to read execution log."
